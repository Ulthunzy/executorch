cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(et_infer_jni LANGUAGES C CXX)
endif()
get_filename_component(_repo "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(MOE_BUILD_DIR "${_repo}/build/moe/qwen35-35B-A3B" CACHE PATH "MoE build.sh output")
if(NOT EXISTS "${MOE_BUILD_DIR}/TARGET.json")
  message(FATAL_ERROR "Missing TARGET.json; run the MoE build.sh first")
endif()
file(READ "${MOE_BUILD_DIR}/TARGET.json" _moe_target)
string(JSON _moe_layout ERROR_VARIABLE _moe_layout_error GET "${_moe_target}" runtime_layout)
if(_moe_layout_error OR NOT _moe_layout STREQUAL "two_libraries")
  message(FATAL_ERROR "Rebuild the MoE runtime with libet_infer.so and libet_qnnop.so")
endif()
string(JSON _moe_dsp_arch GET "${_moe_target}" target dsp_arch)
string(REGEX REPLACE "^v" "" _moe_dsp_arch "${_moe_dsp_arch}")
set(MOE_DSP_ARCH "${_moe_dsp_arch}" CACHE STRING "HTP ISA" FORCE)
string(JSON _moe_native_profile GET "${_moe_target}" native_profile)
set(MOE_NATIVE_PROFILE "${_moe_native_profile}" CACHE STRING "Native profiling" FORCE)
string(JSON _moe_idle_power ERROR_VARIABLE _moe_idle_power_error GET "${_moe_target}" idle_power_management)
if(_moe_idle_power_error)
  set(_moe_idle_power OFF)
endif()
set(MOE_ENABLE_IDLE_POWER_MANAGEMENT "${_moe_idle_power}" CACHE BOOL "Experimental idle power management" FORCE)
set(MOE_NATIVE_BUILD "${MOE_BUILD_DIR}" CACHE PATH "Generated FastRPC interfaces and DSP library" FORCE)
set(MOE_BUILD_TOOLS OFF CACHE BOOL "Build CLI runners and export tools" FORCE)
add_subdirectory("${_repo}/examples/qualcomm/oss_scripts/moe" "${CMAKE_CURRENT_BINARY_DIR}/moe-runtime")
